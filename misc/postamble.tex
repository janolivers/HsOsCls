% Verzeichnisse
%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Einträge im Inhaltsverzeichnis werden vor dem jeweiligen Verzeichnis gesetzt,
% damit Seitenzahl und Link auf dessen erste Seite zeigen.
\cleardoublepage
\pdfbookmark[chapter]{Verzeichnisse}{verzeichnisse}

\iftrue
	\phantomsection
	\addcontentsline{toc}{chapter}{\listfigurename}
	\bookmarksetupnext{level=section}
	\listoffigures
	\clearpage
\fi

\iftrue
	\phantomsection
	\addcontentsline{toc}{chapter}{\listtablename}
	\bookmarksetupnext{level=section}
	\listoftables
	\clearpage
\fi

\iftrue
	\phantomsection
	\addcontentsline{toc}{chapter}{\lstlistlistingname}
	\bookmarksetupnext{level=section}
	\lstlistoflistings
	\clearpage
\fi

\iftrue
	% Glossar
	%%%%%%%%%%%%%%%%%%%%%%%%%%%
	\printunsrtglossary[type=main,style=altlist]
	\clearpage
\fi

	\listoftodos
\fi
